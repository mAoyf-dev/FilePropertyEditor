cmake_minimum_required(VERSION 3.20)
project(FilePropertyEditor VERSION 2.1.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0601 NTDDI_VERSION=0x06010000)
add_compile_options(/utf-8 /W4 /permissive- /EHsc)
add_library(properties STATIC src/core.cpp src/inspect.cpp src/edit.cpp src/version.cpp src/export.cpp src/update.cpp)
target_include_directories(properties PUBLIC src)
target_link_libraries(properties PUBLIC comctl32 comdlg32 shell32 shlwapi propsys ole32 oleaut32 uuid version advapi32 wintrust crypt32 msxml6 winhttp)
add_executable(FilePropertyEditor WIN32 src/app.cpp src/app.rc)
target_link_libraries(FilePropertyEditor PRIVATE properties)
target_link_options(FilePropertyEditor PRIVATE /MANIFEST:NO /SUBSYSTEM:WINDOWS,6.01)
add_executable(property_tests tests/tests.cpp)
target_link_libraries(property_tests PRIVATE properties)
enable_testing()
add_test(NAME property_tests COMMAND property_tests)
set_tests_properties(property_tests PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")

add_executable(update_tests tests/update_tests.cpp)
target_link_libraries(update_tests PRIVATE properties)
add_test(NAME update_tests COMMAND update_tests)
set_tests_properties(update_tests PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
